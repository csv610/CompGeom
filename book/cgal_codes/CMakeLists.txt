cmake_minimum_required(VERSION 3.20)
project(compgeom_cgal_examples LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

find_package(CGAL CONFIG REQUIRED)

function(add_cgal_example target source)
  add_executable(${target} ${source})
  target_link_libraries(${target} PRIVATE CGAL::CGAL)
  target_compile_features(${target} PRIVATE cxx_std_17)
endfunction()

add_cgal_example(convex_hull convex_hull.cpp)
add_cgal_example(segment_intersection segment_intersection.cpp)
add_cgal_example(delaunay_triangulation delaunay_triangulation.cpp)
add_cgal_example(polygon_boolean polygon_boolean.cpp)
add_cgal_example(arrangement_point_location arrangement_point_location.cpp)
add_cgal_example(delaunay_mesher delaunay_mesher.cpp)
add_cgal_example(voronoi voronoi.cpp)
add_cgal_example(generate_points generate_points.cpp)
add_cgal_example(verify_convex_hull verify_convex_hull.cpp)
add_cgal_example(verify_voronoi verify_voronoi.cpp)
add_cgal_example(verify_delaunay verify_delaunay.cpp)
add_cgal_example(benchmark_convex_hull benchmark_convex_hull.cpp)
add_cgal_example(benchmark_delaunay_mesh benchmark_delaunay_mesh.cpp)
add_cgal_example(benchmark_voronoi benchmark_voronoi.cpp)
add_cgal_example(constrained_delaunay_mesh constrained_delaunay_mesh.cpp)
add_cgal_example(constrained_voronoi constrained_voronoi.cpp)
